```py
sir = input()

numar = 0
for litera in sir:
    if litera == 'a' or litera == 'e' or litera == 'i' or litera == 'o' or litera == 'u':
        numar = numar + 1
    elif litera == 'A' or litera == 'E' or litera == 'I' or litera == 'O' or litera == 'U':
        numar = numar + 1

print(numar)
```
