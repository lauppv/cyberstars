public class Main {
    public static int impartePrada(int prada, int membri) {
        return prada / membri;
    }

    public static void main(String[] args) {
        int prada = 10000;
        int membri = 4;

        System.out.println(prada + " / " + membri + " = " + impartePrada(prada, membri));
    }
}
